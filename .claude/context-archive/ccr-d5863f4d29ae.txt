//! BSP (Binary Space Partitioning) file loader for Quake 2 `.bsp` maps.
//!
//! Parses the IBSP v38 format used by Quake 2 into typed Rust structs.
//! All on-disk values are little-endian.
//!
//! Reference: `common/header/files.h` in the Quake 2 source (struct definitions),
//! and `gl3/gl3_model.c` / `files/models.c` for the loading logic.

use q2_common::bsp_file::{
    BspHeader, Lump, LUMP_EDGES, LUMP_FACES, LUMP_LEAFFACES, LUMP_LIGHTING, LUMP_SURFEDGES,
    LUMP_VERTEXES, LUMP_VISIBILITY,
};
use q2_common::error::{Q2Error, Q2Result};
use q2_shared::types::Vec3f;

// On-disk struct sizes (bytes). Used for lump validation.
const DISK_VERTEX_SIZE: usize = 12; // 3 * f32
const DISK_EDGE_SIZE: usize = 4; // 2 * u16
const DISK_FACE_SIZE: usize = 20; // u16 + i16 + i32 + i16 + i16 + 4*u8 + i32

// ---------------------------------------------------------------------------
// Parsed BSP types
// ---------------------------------------------------------------------------

/// A vertex (3D position).
#[derive(Debug, Clone)]
pub struct BspVertex {
    pub position: Vec3f,
}

/// An edge — two vertex indices.
#[derive(Debug, Clone)]
pub struct BspEdge {
    pub v: [u16; 2],
}

/// Texture-mapping info for a face.
///
/// `vecs` encodes the texture projection: for each axis (s, t) there are four
/// floats `[x, y, z, offset]` such that `s = dot(position, xyz) + offset`.
#[derive(Debug, Clone)]
pub struct BspTexInfo {
    /// `[s/t][xyz offset]` — texture projection vectors.
    pub vecs: [[f32; 4]; 2],
    /// Surface flags (`SURF_*`).
    pub flags: i32,
    /// Light emission value.
    pub value: i32,
    /// Texture name (e.g. `"e1u1/floor1_1"`), up to 32 chars.
    pub texture: String,
    /// Next texinfo in animation chain, or -1 for none.
    pub next_texinfo: i32,
}

/// A face (polygon on a BSP plane).
#[derive(Debug, Clone)]
pub struct BspFace {
    pub plane_idx: u16,
    /// 0 = front side, 1 = back side of the plane.
    pub side: u16,
    /// Index into the surface-edges array.
    pub first_edge: i32,
    /// Number of edges (and thus vertices) for this face.
    pub num_edges: i16,
    /// Index into the texinfo array.
    pub texinfo_idx: i16,
    /// Lightmap styles (up to 4).
    pub styles: [u8; 4],
    /// Byte offset into lightmap data, or -1 for no lightmap.
    pub light_ofs: i32,
}

/// A BSP plane (splitting plane of the tree).
#[derive(Debug, Clone)]
pub struct BspPlane {
    pub normal: Vec3f,
    pub dist: f32,
    /// Plane type: 0-2 = axial (X/Y/Z), 3-5 = nearest axial.
    pub plane_type: i32,
}

/// A BSP internal node.
#[derive(Debug, Clone)]
pub struct BspNode {
    pub plane_idx: i32,
    /// Children: positive = node index, negative = -(leaf+1).
    pub children: [i32; 2],
    pub mins: [i16; 3],
    pub maxs: [i16; 3],
    pub first_face: u16,
    pub num_faces: u16,
}

/// A BSP leaf (terminal node).
#[derive(Debug, Clone)]
pub struct BspLeaf {
    /// OR of all brush contents in this leaf.
    pub contents: i32,
    /// Visibility cluster, or -1.
    pub cluster: i16,
    /// Area index.
    pub area: i16,
    pub first_leaf_face: u16,
    pub num_leaf_faces: u16,
    pub first_leaf_brush: u16,
    pub num_leaf_brushes: u16,
}

/// A submodel (the world model is index 0, doors/platforms are 1+).
#[derive(Debug, Clone)]
pub struct BspModel {
    pub mins: Vec3f,
    pub maxs: Vec3f,
    pub origin: Vec3f,
    pub head_node: i32,
    pub first_face: i32,
    pub num_faces: i32,
}

/// Complete parsed BSP data for rendering.
#[derive(Debug)]
pub struct BspData {
    pub planes: Vec<BspPlane>,
    pub vertices: Vec<BspVertex>,
    pub visibility: Vec<u8>,
    pub nodes: Vec<BspNode>,
    pub texinfo: Vec<BspTexInfo>,
    pub faces: Vec<BspFace>,
    pub lightmap_data: Vec<u8>,
    pub leafs: Vec<BspLeaf>,
    pub leaf_faces: Vec<u16>,
    pub edges: Vec<BspEdge>,
    pub surface_edges: Vec<i32>,
    pub models: Vec<BspModel>,
}

use q2_common::binary::{try_read_f32, try_read_i16, try_read_i32, try_read_u16};

/// Read a `Vec3f` (3 consecutive LE f32s), returning an error on out-of-bounds.
fn try_read_vec3(data: &[u8], off: usize) -> Q2Result<Vec3f> {
    Ok(Vec3f::new(
        try_read_f32(data, off)?,
        try_read_f32(data, off + 4)?,
        try_read_f32(data, off + 8)?,
    ))
}

// ---------------------------------------------------------------------------
// Lump validation helper
// ---------------------------------------------------------------------------

/// Validate that a lump fits within the file and has a size that is a multiple
/// of the given struct size. Returns (offset, count).
fn validate_lump(
    data: &[u8],
    lump: &Lump,
    struct_size: usize,
    name: &str,
) -> Q2Result<(usize, usize)> {
    let off = lump.offset;
    let len = lump.length;

    if off.checked_add(len).is_none() || off + len > data.len() {
        return Err(Q2Error::Drop(format!(
            "BSP {name} lump extends past end of file"
        )));
    }

    if struct_size > 0 && !len.is_multiple_of(struct_size) {
        return Err(Q2Error::Drop(format!(
            "BSP {name} lump has invalid size (not a multiple of {struct_size})"
        )));
    }

    let count = len.checked_div(struct_size).unwrap_or(0);
    Ok((off, count))
}

// ---------------------------------------------------------------------------
// Individual lump loaders
// ---------------------------------------------------------------------------

fn load_planes(header: &BspHeader, data: &[u8]) -> Q2Result<Vec<BspPlane>> {
    Ok(header
        .planes(data)?
        .into_iter()
        .map(|p| BspPlane {
            normal: p.normal,
            dist: p.dist,
            plane_type: p.plane_type,
        })
        .collect())
}

fn load_vertices(data: &[u8], lump: &Lump) -> Q2Result<Vec<BspVertex>> {
    let (off, count) = validate_lump(data, lump, DISK_VERTEX_SIZE, "vertices")?;
    let mut verts = Vec::with_capacity(count);
    for i in 0..count {
        let base = off + i * DISK_VERTEX_SIZE;
        verts.push(BspVertex {
            position: try_read_vec3(data, base)?,
        });
    }
    Ok(verts)
}

fn load_visibility(data: &[u8], lump: &Lump) -> Q2Result<Vec<u8>> {
    let (off, len) = validate_lump(data, lump, 1, "visibility")?;
    Ok(data[off..off + len].to_vec())
}

fn load_nodes(header: &BspHeader, data: &[u8]) -> Q2Result<Vec<BspNode>> {
    Ok(header
        .nodes(data)?
        .into_iter()
        .map(|n| BspNode {
            plane_idx: n.plane_num,
            children: n.children,
            mins: n.mins,
            maxs: n.maxs,
            first_face: n.first_face,
            num_faces: n.num_faces,
        })
        .collect())
}

fn load_texinfo(header: &BspHeader, data: &[u8]) -> Q2Result<Vec<BspTexInfo>> {
    Ok(header
        .texinfo(data)?
        .into_iter()
        .map(|t| BspTexInfo {
            vecs: t.vecs,
            flags: t.flags,
            value: t.value,
            texture: t.texture,
            next_texinfo: t.next_texinfo,
        })
        .collect())
}

fn load_faces(data: &[u8], lump: &Lump) -> Q2Result<Vec<BspFace>> {
    let (off, count) = validate_lump(data, lump, DISK_FACE_SIZE, "faces")?;
    let mut faces = Vec::with_capacity(count);
    for i in 0..count {
        let base = off + i * DISK_FACE_SIZE;
        faces.push(BspFace {
            plane_idx: try_read_u16(data, base)?,
            side: try_read_u16(data, base + 2)?,
            first_edge: try_read_i32(data, base + 4)?,
            num_edges: try_read_i16(data, base + 8)?,
            texinfo_idx: try_read_i16(data, base + 10)?,
            styles: [
                data[base + 12],
                data[base + 13],
                data[base + 14],
                data[base + 15],
            ],
            light_ofs: try_read_i32(data, base + 16)?,
        });
    }
    Ok(faces)
}

fn load_lightmap(data: &[u8], lump: &Lump) -> Q2Result<Vec<u8>> {
    let (off, len) = validate_lump(data, lump, 1, "lightmap")?;
    Ok(data[off..off + len].to_vec())
}

fn load_leafs(header: &BspHeader, data: &[u8]) -> Q2Result<Vec<BspLeaf>> {
    Ok(header
        .leafs(data)?
        .into_iter()
        .map(|l| BspLeaf {
            contents: l.contents,
            cluster: l.cluster,
            area: l.area,
            first_leaf_face: l.first_leaf_face,
            num_leaf_faces: l.num_leaf_faces,
            first_leaf_brush: l.first_leaf_brush,
            num_leaf_brushes: l.num_leaf_brushes,
        })
        .collect())
}

fn load_leaf_faces(data: &[u8], lump: &Lump) -> Q2Result<Vec<u16>> {
    let (off, count) = validate_lump(data, lump, 2, "leaf_faces")?;
    let mut lf = Vec::with_capacity(count);
    for i in 0..count {
        lf.push(try_read_u16(data, off + i * 2)?);
    }
    Ok(lf)
}

fn load_edges(data: &[u8], lump: &Lump) -> Q2Result<Vec<BspEdge>> {
    let (off, count) = validate_lump(data, lump, DISK_EDGE_SIZE, "edges")?;
    let mut edges = Vec::with_capacity(count);
    for i in 0..count {
        let base = off + i * DISK_EDGE_SIZE;
        edges.push(BspEdge {
            v: [try_read_u16(data, base)?, try_read_u16(data, base + 2)?],
        });
    }
    Ok(edges)
}

fn load_surface_edges(data: &[u8], lump: &Lump) -> Q2Result<Vec<i32>> {
    let (off, count) = validate_lump(data, lump, 4, "surface_edges")?;
    let mut se = Vec::with_capacity(count);
    for i in 0..count {
        se.push(try_read_i32(data, off + i * 4)?);
    }
    Ok(se)
}

fn load_models(header: &BspHeader, data: &[u8]) -> Q2Result<Vec<BspModel>> {
    let spread = Vec3f::new(1.0, 1.0, 1.0);
    Ok(header
        .models(data)?
        .into_iter()
        .map(|m| BspModel {
            // Spread the bounds by one unit like gl3_model.c Mod_LoadSubmodels.
            mins: m.mins - spread,
            maxs: m.maxs + spread,
            origin: m.origin,
            head_node: m.head_node,
            first_face: m.first_face,
            num_faces: m.num_faces,
        })
        .collect())
}

// ---------------------------------------------------------------------------
// Main loader
// ---------------------------------------------------------------------------

impl BspData {
    /// Parse BSP data from raw bytes (the contents of a `.bsp` file).
    ///
    /// Validates the header magic and version, then reads each lump into
    /// typed Rust structures. All values are assumed little-endian (x86/wasm).
    pub fn load(data: &[u8]) -> Q2Result<Self> {
        let header = BspHeader::parse(data)?;
        let lumps = &header.lumps;

        // Load each lump
        let planes = load_planes(&header, data)?;
        let vertices = load_vertices(data, &lumps[LUMP_VERTEXES])?;
        let visibility = load_visibility(data, &lumps[LUMP_VISIBILITY])?;
        let nodes = load_nodes(&header, data)?;
        let texinfo = load_texinfo(&header, data)?;
        let faces = load_faces(data, &lumps[LUMP_FACES])?;
        let lightmap_data = load_lightmap(data, &lumps[LUMP_LIGHTING])?;
        let leafs = load_leafs(&header, data)?;
        let leaf_faces = load_leaf_faces(data, &lumps[LUMP_LEAFFACES])?;
        let edges = load_edges(data, &lumps[LUMP_EDGES])?;
        let surface_edges = load_surface_edges(data, &lumps[LUMP_SURFEDGES])?;
        let models = load_models(&header, data)?;

        Ok(BspData {
            planes,
            vertices,
            visibility,
            nodes,
            texinfo,
            faces,
            lightmap_data,
            leafs,
            leaf_faces,
            edges,
            surface_edges,
            models,
        })
    }
}

#[cfg(test)]
mod tests {
    use super::*;
    use q2_common::bsp_file::{BSPVERSION, HEADER_LUMPS, IDBSPHEADER};

    /// Build a minimal valid BSP byte buffer with header + empty lumps.
    fn make_minimal_bsp() -> Vec<u8> {
        let header_size: u32 = (4 + 4 + HEADER_LUMPS * 8) as u32;
        let mut buf = Vec::new();

        // Magic
        buf.extend_from_slice(&IDBSPHEADER.to_le_bytes());
        // Version
        buf.extend_from_slice(&BSPVERSION.to_le_bytes());

        // 19 lumps, all pointing to offset=header_size, length=0
        for _ in 0..HEADER_LUMPS {
            buf.extend_from_slice(&header_size.to_le_bytes()); // offset
            buf.extend_from_slice(&0u32.to_le_bytes()); // length
        }

        buf
    }

    #[test]
    fn bsp_header_parse() {
        let data = make_minimal_bsp();
        let bsp = BspData::load(&data).expect("should parse minimal BSP");
        assert!(bsp.vertices.is_empty());
        assert!(bsp.faces.is_empty());
        assert!(bsp.planes.is_empty());
        assert!(bsp.nodes.is_empty());
        assert!(bsp.leafs.is_empty());
        assert!(bsp.edges.is_empty());
        assert!(bsp.models.is_empty());
        assert!(bsp.texinfo.is_empty());
    }

    #[test]
    fn bsp_reject_invalid() {
        // Too short
        assert!(BspData::load(&[0u8; 10]).is_err());

        // Wrong magic
        let mut bad_magic = make_minimal_bsp();
        bad_magic[0] = 0xFF;
        assert!(BspData::load(&bad_magic).is_err());

        // Wrong version
        let mut bad_version = make_minimal_bsp();
        bad_version[4] = 99;
        bad_version[5] = 0;
        bad_version[6] = 0;
        bad_version[7] = 0;
        assert!(BspData::load(&bad_version).is_err());
    }

    #[test]
    fn bsp_parse_vertices() {
        let header_size = (4 + 4 + HEADER_LUMPS * 8) as u32;
        let mut buf = Vec::new();

        // Magic + version
        buf.extend_from_slice(&IDBSPHEADER.to_le_bytes());
        buf.extend_from_slice(&BSPVERSION.to_le_bytes());

        // Vertex data will live right after header.
        // We place 2 vertices (24 bytes) for the vertex lump and leave everything else empty.
        let vert_offset = header_size;
        let vert_length: u32 = 2 * DISK_VERTEX_SIZE as u32;

        for i in 0..HEADER_LUMPS {
            if i == LUMP_VERTEXES {
                buf.extend_from_slice(&vert_offset.to_le_bytes());
                buf.extend_from_slice(&vert_length.to_le_bytes());
            } else {
                // Point to after vert data, length 0
                buf.extend_from_slice(&(vert_offset + vert_length).to_le_bytes());
                buf.extend_from_slice(&0u32.to_le_bytes());
            }
        }

        // Write 2 vertices
        // Vertex 0: (1.0, 2.0, 3.0)
        buf.extend_from_slice(&1.0f32.to_le_bytes());
        buf.extend_from_slice(&2.0f32.to_le_bytes());
        buf.extend_from_slice(&3.0f32.to_le_bytes());
        // Vertex 1: (4.0, 5.0, 6.0)
        buf.extend_from_slice(&4.0f32.to_le_bytes());
        buf.extend_from_slice(&5.0f32.to_le_bytes());
        buf.extend_from_slice(&6.0f32.to_le_bytes());

        let bsp = BspData::load(&buf).expect("should parse BSP with vertices");
        assert_eq!(bsp.vertices.len(), 2);
        assert_eq!(bsp.vertices[0].position, Vec3f::new(1.0, 2.0, 3.0));
        assert_eq!(bsp.vertices[1].position, Vec3f::new(4.0, 5.0, 6.0));
    }

    #[test]
    fn submodel_bounds_are_spread_by_one_unit() {
        // gl3_model.c Mod_LoadSubmodels: "spread the mins / maxs by a pixel".
        let mut record = Vec::new();
        for f in [0.0f32, 0.0, 0.0, 10.0, 10.0, 10.0, 5.0, 5.0, 5.0] {
            record.extend_from_slice(&f.to_le_bytes());
        }
        for i in [0i32, 0, 1] {
            record.extend_from_slice(&i.to_le_bytes());
        }
        let mut data = make_minimal_bsp();
        let at = 8 + q2_common::bsp_file::LUMP_MODELS * 8;
        data[at + 4..at + 8].copy_from_slice(&(record.len() as u32).to_le_bytes());
        data.extend_from_slice(&record);
        let models = BspData::load(&data).unwrap().models;
        assert_eq!(models[0].mins, Vec3f::new(-1.0, -1.0, -1.0));
        assert_eq!(models[0].maxs, Vec3f::new(11.0, 11.0, 11.0));
        assert_eq!(models[0].origin, Vec3f::new(5.0, 5.0, 5.0));
    }

    #[test]
    fn raw_lumps_whose_end_overflows_are_rejected() {
        // offset + length wraps a 32-bit usize (wasm32) to a small value.
        let data = make_minimal_bsp();
        let lump = Lump {
            offset: u32::MAX as usize,
            length: 2,
        };
        assert!(load_visibility(&data, &lump).is_err());
        assert!(load_lightmap(&data, &lump).is_err());
    }

    #[test]
    fn bsp_reject_misaligned_lump() {
        let header_size = (4 + 4 + HEADER_LUMPS * 8) as u32;
        let mut buf = Vec::new();

        buf.extend_from_slice(&IDBSPHEADER.to_le_bytes());
        buf.extend_from_slice(&BSPVERSION.to_le_bytes());

        // Vertex lump with length that is not a multiple of 12
        for i in 0..HEADER_LUMPS {
            if i == LUMP_VERTEXES {
                buf.extend_from_slice(&header_size.to_le_bytes());
                buf.extend_from_slice(&7u32.to_le_bytes()); // 7 is not a multiple of 12
            } else {
                buf.extend_from_slice(&header_size.to_le_bytes());
                buf.extend_from_slice(&0u32.to_le_bytes());
            }
        }

        // Pad some data so the offset is valid
        buf.extend_from_slice(&[0u8; 12]);

        assert!(BspData::load(&buf).is_err());
    }
}
//! Parsers for Quake 2 asset formats the renderer consumes: PCX, WAL, MD2, SP2.
//!
//! All parsers take a byte slice, never read out of bounds, and report
//! malformed data as `Q2Error::Drop` (C `Com_Error(ERR_DROP)` / failed load).

mod anorms;
pub mod md2;
pub mod palette;
pub mod pcx;
pub mod sp2;
pub mod wal;

use q2_common::error::{Q2Error, Q2Result};

/// Build the error every parser returns for malformed input.
pub(crate) fn malformed(what: &str, why: impl std::fmt::Display) -> Q2Error {
    Q2Error::Drop(format!("{what}: {why}"))
}

/// Decode a NUL-terminated string from a fixed-size on-disk field.
pub(crate) fn fixed_cstring(bytes: &[u8]) -> String {
    let end = bytes.iter().position(|&b| b == 0).unwrap_or(bytes.len());
    String::from_utf8_lossy(&bytes[..end]).into_owned()
}

/// Convert an on-disk signed count or offset to `usize`, rejecting negatives.
pub(crate) fn non_negative(what: &str, field: &str, value: i32) -> Q2Result<usize> {
    usize::try_from(value).map_err(|_| malformed(what, format_args!("negative {field} {value}")))
}

/// Borrow `len` bytes at `off`, failing instead of panicking when out of range.
pub(crate) fn slice_at<'a>(
    what: &str,
    data: &'a [u8],
    off: usize,
    len: usize,
) -> Q2Result<&'a [u8]> {
    off.checked_add(len)
        .and_then(|end| data.get(off..end))
        .ok_or_else(|| {
            malformed(
                what,
                format_args!("{len} bytes at offset {off} exceed file"),
            )
        })
}

#[cfg(test)]
mod real_assets {
    use super::md2::{load_md2, VERTEX_NORMALS};
    use super::palette::Palette;
    use super::pcx::load_pcx;
    use super::sp2::load_sp2;
    use super::wal::load_wal;
    use q2_common::filesystem::{FileSystem, Pack};
    use std::path::PathBuf;

    /// Open `pak0.pak` from `Q2_GAMEDATA_DIR` or `<repo>/gamedata`; `None` when absent.
    fn open_baseq2() -> Option<FileSystem> {
        let root = std::env::var_os("Q2_GAMEDATA_DIR")
            .map(PathBuf::from)
            .unwrap_or_else(|| PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("../../gamedata"));
        let pak_path = root.join("baseq2/pak0.pak");
        if !pak_path.is_file() {
            return None;
        }
        let mut filesystem = FileSystem::new("baseq2");
        filesystem.add_pack(Pack::load(&pak_path).expect("pak0.pak parses"));
        Some(filesystem)
    }

    #[test]
    fn parses_stock_assets_from_pak0() {
        let Some(filesystem) = open_baseq2() else {
            return;
        };

        let colormap = filesystem.load_file("pics/colormap.pcx").unwrap();
        let palette = Palette::from_pcx(&colormap).unwrap();
        assert_eq!(palette.rgba(255)[3], 0);
        let image = load_pcx(&colormap).unwrap();
        assert_eq!(image.pixels.len(), image.width * image.height);

        let wal_names = filesystem.list_files("wal");
        let wal_name = wal_names
            .iter()
            .find(|name| name.starts_with("textures/"))
            .expect("pak0 holds textures");
        let texture = load_wal(&filesystem.load_file(wal_name).unwrap()).unwrap();
        assert_eq!(texture.pixels.len(), texture.width * texture.height);

        let model = load_md2(
            &filesystem
                .load_file("models/monsters/soldier/tris.md2")
                .unwrap(),
        )
        .unwrap();
        assert!(!model.frames.is_empty() && !model.triangles.is_empty());
        let normal_limit = u8::try_from(VERTEX_NORMALS.len()).unwrap_or(u8::MAX);
        assert!(model.frames[0]
            .verts
            .iter()
            .all(|vertex| vertex.normal_index < normal_limit));

        let sprite_names = filesystem.list_files("sp2");
        let sprite_name = sprite_names.first().expect("pak0 holds sprites");
        assert!(!load_sp2(&filesystem.load_file(sprite_name).unwrap())
            .unwrap()
            .frames
            .is_empty());
    }
}
//! 256-colour palette from `pics/colormap.pcx` (C `R_GetPalette` / `GetPCXPalette`).

use super::{malformed, slice_at};
use q2_common::error::Q2Result;

/// Palette index that is transparent in 32-bit conversion (`d_8to24table[255]`).
pub const TRANSPARENT_INDEX: u8 = 255;

const PALETTE_BYTES: usize = 768;

/// 256 RGB entries.
#[derive(Debug, Clone, PartialEq, Eq)]
pub struct Palette(pub [[u8; 3]; 256]);

impl Palette {
    /// Read the 768-byte palette stored at the very end of a PCX file.
    pub fn from_pcx(pcx: &[u8]) -> Q2Result<Palette> {
        const WHAT: &str = "palette";
        // C GetPCXPalette requires len > 768 so the image header precedes it.
        if pcx.len() <= PALETTE_BYTES {
            return Err(malformed(WHAT, "pcx too small to hold a palette"));
        }
        let raw = slice_at(WHAT, pcx, pcx.len() - PALETTE_BYTES, PALETTE_BYTES)?;
        Ok(Palette(rgb_entries(raw)))
    }

    /// RGBA for an index; index 255 has alpha 0, all others 255.
    pub fn rgba(&self, index: u8) -> [u8; 4] {
        let [r, g, b] = self.0[usize::from(index)];
        let alpha = if index == TRANSPARENT_INDEX { 0 } else { 255 };
        [r, g, b, alpha]
    }
}

/// Split exactly 768 bytes into 256 RGB triples.
pub(crate) fn rgb_entries(raw: &[u8]) -> [[u8; 3]; 256] {
    let mut entries = [[0u8; 3]; 256];
    for (entry, rgb) in entries.iter_mut().zip(raw.chunks_exact(3)) {
        entry.copy_from_slice(rgb);
    }
    entries
}

#[cfg(test)]
mod tests {
    use super::*;

    fn pcx_with_palette(prefix_len: usize) -> Vec<u8> {
        let mut bytes = vec![0xAAu8; prefix_len];
        for i in 0..=255u8 {
            bytes.extend_from_slice(&[i, i.wrapping_add(1), i.wrapping_add(2)]);
        }
        bytes
    }

    #[test]
    fn reads_palette_from_file_tail() {
        let palette = Palette::from_pcx(&pcx_with_palette(10)).unwrap();
        assert_eq!(palette.0[0], [0, 1, 2]);
        assert_eq!(palette.0[200], [200, 201, 202]);
    }

    #[test]
    fn rejects_file_no_larger_than_palette() {
        assert!(Palette::from_pcx(&pcx_with_palette(0)).is_err());
        assert!(Palette::from_pcx(&[]).is_err());
    }

    #[test]
    fn index_255_is_transparent_others_opaque() {
        let palette = Palette::from_pcx(&pcx_with_palette(1)).unwrap();
        assert_eq!(palette.rgba(255), [255, 0, 1, 0]);
        assert_eq!(palette.rgba(7), [7, 8, 9, 255]);
    }
}
//! Quake 2 `.wal` texture loader (C `LoadWal` in `refresh/files/wal.c`).

use super::{fixed_cstring, malformed};
use q2_common::binary::{try_read_i32, try_read_u32};
use q2_common::error::Q2Result;

const WHAT: &str = "wal";
/// `sizeof(miptex_t)`: 32 name + 2*4 size + 4*4 offsets + 32 animname + 3*4.
const HEADER_SIZE: usize = 100;

/// WAL texture; only mip level 0 is kept (the GPU generates the rest).
#[derive(Debug, Clone)]
pub struct WalTexture {
    pub name: String,
    pub width: usize,
    pub height: usize,
    /// Palette indices of mip 0, `width * height` bytes.
    pub pixels: Vec<u8>,
    /// Next texture in the animation chain, empty for none.
    pub animname: String,
    pub flags: i32,
    pub contents: i32,
    pub value: i32,
}

pub fn load_wal(data: &[u8]) -> Q2Result<WalTexture> {
    if data.len() < HEADER_SIZE {
        return Err(malformed(WHAT, "small header"));
    }
    let width = try_read_u32(data, 32)?;
    let height = try_read_u32(data, 36)?;
    let offset = try_read_u32(data, 40)?;

    // C reads these as signed ints, so values above i32::MAX count as negative.
    let (width, height, offset) = match (
        i32::try_from(width),
        i32::try_from(height),
        i32::try_from(offset),
    ) {
        (Ok(w), Ok(h), Ok(o)) if w > 0 && h > 0 && o > 0 => (w, h, o),
        _ => return Err(malformed(WHAT, "bad dimensions or offset")),
    };
    let width = usize::try_from(width).map_err(|e| malformed(WHAT, e))?;
    let height = usize::try_from(height).map_err(|e| malformed(WHAT, e))?;
    let offset = usize::try_from(offset).map_err(|e| malformed(WHAT, e))?;

    let pixel_count = width
        .checked_mul(height)
        .ok_or_else(|| malformed(WHAT, "dimensions overflow"))?;
    let end = offset
        .checked_add(pixel_count)
        .filter(|&end| end <= data.len())
        .ok_or_else(|| malformed(WHAT, "small body"))?;

    Ok(WalTexture {
        name: fixed_cstring(&data[0..32]),
        width,
        height,
        pixels: data[offset..end].to_vec(),
        animname: fixed_cstring(&data[56..88]),
        flags: try_read_i32(data, 88)?,
        contents: try_read_i32(data, 92)?,
        value: try_read_i32(data, 96)?,
    })
}

#[cfg(test)]
mod tests {
    use super::*;

    fn wal(width: u32, height: u32, offset: u32, body: &[u8]) -> Vec<u8> {
        let mut bytes = vec![0u8; HEADER_SIZE];
        bytes[..9].copy_from_slice(b"e1u1/test");
        bytes[32..36].copy_from_slice(&width.to_le_bytes());
        bytes[36..40].copy_from_slice(&height.to_le_bytes());
        bytes[40..44].copy_from_slice(&offset.to_le_bytes());
        bytes[56..61].copy_from_slice(b"next1");
        bytes[88..92].copy_from_slice(&5i32.to_le_bytes());
        bytes[92..96].copy_from_slice(&(-2i32).to_le_bytes());
        bytes[96..100].copy_from_slice(&7i32.to_le_bytes());
        bytes.extend_from_slice(body);
        bytes
    }

    #[test]
    fn parses_header_and_first_mip() {
        let texture = load_wal(&wal(2, 2, 100, &[1, 2, 3, 4, 9, 9])).unwrap();
        assert_eq!(texture.name, "e1u1/test");
        assert_eq!((texture.width, texture.height), (2, 2));
        assert_eq!(texture.pixels, vec![1, 2, 3, 4]);
        assert_eq!(texture.animname, "next1");
        assert_eq!((texture.flags, texture.contents, texture.value), (5, -2, 7));
    }

    #[test]
    fn rejects_small_header() {
        assert!(load_wal(&[0u8; 99]).is_err());
    }

    #[test]
    fn rejects_zero_dimensions_and_zero_offset() {
        assert!(load_wal(&wal(0, 2, 100, &[0; 4])).is_err());
        assert!(load_wal(&wal(2, 0, 100, &[0; 4])).is_err());
        assert!(load_wal(&wal(2, 2, 0, &[0; 4])).is_err());
    }

    #[test]
    fn rejects_body_shorter_than_pixels() {
        assert!(load_wal(&wal(2, 2, 100, &[0; 3])).is_err());
    }

    #[test]
    fn rejects_offset_past_end_and_overflowing_sizes() {
        assert!(load_wal(&wal(1, 1, 5000, &[0; 4])).is_err());
        assert!(load_wal(&wal(u32::MAX, u32::MAX, 100, &[0; 4])).is_err());
        assert!(load_wal(&wal(0x7fff_ffff, 0x7fff_ffff, 100, &[0; 4])).is_err());
        assert!(load_wal(&wal(2, 2, u32::MAX, &[0; 4])).is_err());
    }
}
